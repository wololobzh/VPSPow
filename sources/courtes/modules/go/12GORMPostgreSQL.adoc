== Configuration GORM

Installation et configuration initiale :

[source,go]
----
// Installation
go get -u gorm.io/gorm
go get -u gorm.io/driver/postgres

// Configuration
type Config struct {
    Host     string
    Port     string
    User     string
    Password string
    DBName   string
    SSLMode  string
}

func NewDatabase(config *Config) (*gorm.DB, error) {
    dsn := fmt.Sprintf("host=%s port=%s user=%s password=%s dbname=%s sslmode=%s",
        config.Host, config.Port, config.User, 
        config.Password, config.DBName, config.SSLMode)
        
    db, err := gorm.Open(postgres.Open(dsn), &gorm.Config{})
    if err != nil {
        return nil, err
    }
    
    return db, nil
}
----

[NOTE.speaker]
--
GORM est un ORM populaire pour Go.
La configuration doit être sécurisée (variables d'environnement).
Important de gérer les connexions proprement.
Les options de configuration permettent d'optimiser les performances.
--

== Migrations

Gestion des migrations avec GORM :

[source,go]
----
func Migrate(db *gorm.DB) error {
    // Auto-migration
    err := db.AutoMigrate(
        &User{},
        &Post{},
        &Comment{},
    )
    if err != nil {
        return fmt.Errorf("erreur migration: %w", err)
    }
    
    // Migration manuelle si nécessaire
    err = db.Exec(`
        CREATE INDEX IF NOT EXISTS idx_users_email 
        ON users(email);
    `).Error
    
    return err
}
----

[NOTE.speaker]
--
AutoMigrate est pratique mais attention en production.
Les migrations manuelles offrent plus de contrôle.
Important de versionner les migrations.
Penser aux rollbacks.
--

== Relations GORM

Types de relations supportées :

* Has One
* Has Many
* Belongs To
* Many To Many

[source,go]
----
type User struct {
    gorm.Model
    Name      string
    Profile   Profile     `gorm:"foreignKey:UserID"`  // Has One
    Posts     []Post      `gorm:"foreignKey:UserID"`  // Has Many
    Languages []Language  `gorm:"many2many:user_languages"` // Many To Many
}

type Profile struct {
    gorm.Model
    UserID    uint
    Bio       string
}
----

[NOTE.speaker]
--
Les relations sont définies via des tags.
GORM gère automatiquement les clés étrangères.
Possibilité de personnaliser les jointures.
Attention aux chargements eager vs lazy.
--

```go
package repository

import (
    "gorm.io/gorm"
    "myapp/models"
)

type UserRepository struct {
    db *gorm.DB
}

func NewUserRepository(db *gorm.DB) *UserRepository {
    return &UserRepository{db: db}
}

func (r *UserRepository) Create(user *models.User) error {
    return r.db.Create(user).Error
}

func (r *UserRepository) FindByID(id uint) (*models.User, error) {
    var user models.User
    err := r.db.Preload("Posts").First(&user, id).Error
    if err != nil {
        return nil, err
    }
    return &user, nil
}

func (r *UserRepository) Update(user *models.User) error {
    return r.db.Save(user).Error
}

func (r *UserRepository) Delete(id uint) error {
    return r.db.Delete(&models.User{}, id).Error
}

func (r *UserRepository) List(page, size int) ([]models.User, error) {
    var users []models.User
    err := r.db.Offset((page - 1) * size).
        Limit(size).
        Preload("Posts").
        Find(&users).Error
    return users, err
}
```

== Transactions

Gestion des transactions avec GORM :

[source,go]
----
func (r *UserRepository) CreateWithPosts(user *models.User) error {
    return r.db.Transaction(func(tx *gorm.DB) error {
        // Créer l'utilisateur
        if err := tx.Create(user).Error; err != nil {
            return err
        }
        
        // Créer les posts
        for _, post := range user.Posts {
            post.UserID = user.ID
            if err := tx.Create(&post).Error; err != nil {
                return err
            }
        }
        
        return nil
    })
}
----

[NOTE.speaker]
--
Les transactions garantissent l'intégrité des données.
GORM gère automatiquement commit/rollback.
Important pour les opérations complexes.
Attention aux deadlocks.
--
